def tap(v, **kwargs):
    """
    画面上の指定された位置またはゲームオブジェクトをタップします。

    使用例:
    tap(path: str, duration: float=0.0)
    path: ゲームオブジェクトのパス
    duration: タップの持続時間（秒）

    tap(pos: tuple[float, float], duration: float=0.0)
    pos: 画面上の座標
    duration: タップの持続時間（秒）

    Args:
        v: ゲームオブジェクトのパス文字列または座標タプル(x, y)
        **kwargs: 追加のパラメータ（duration など）
    """
    Context.device.tap(v, **kwargs)

def swipe(v1, v2=None, **kwargs):
    """
    画面上の2つの指定位置間またはゲームオブジェクト間でスワイプします。

    使用例:
    swipe(path1: str, path2: str, duration: float=0.0)
    path1: 最初のゲームオブジェクトのパス
    path2: 2番目のゲームオブジェクトのパス
    duration: スワイプの持続時間（秒）

    swipe(path1: str, pos2: tuple[float, float], duration: float=0.0)
    path1: 最初のゲームオブジェクトのパス
    pos2: 2番目の位置座標
    duration: スワイプの持続時間（秒）

    swipe(path1: str, vector: tuple[float, float], duration: float=0.0)
    path1: 最初のゲームオブジェクトのパス
    vector: スワイプの方向ベクトル
    duration: スワイプの持続時間（秒）

    swipe(pos1: tuple[float, float], pos2: tuple[float, float], duration: float=0.0)
    pos1: 開始位置の座標
    pos2: 終了位置の座標
    duration: スワイプの持続時間（秒）

    swipe(pos1: tuple[float, float], vector: tuple[float, float], duration: float=0.0)
    pos1: 開始位置の座標
    vector: スワイプの方向ベクトル
    duration: スワイプの持続時間（秒）

    Args:
        v1: 開始位置（ゲームオブジェクトのパスまたは座標タプル）
        v2: 終了位置または方向ベクトル（デフォルト: None）
        **kwargs: 追加のパラメータ（duration など）
    """
    Context.device.swipe(v1, v2, **kwargs)

def sleep(seconds, check_interval=0.1):
    """
    指定された秒数だけ実行を一時停止します。

    Args:
        seconds (float): 停止する秒数
        check_interval (float, optional): チェック間隔（秒）。デフォルトは0.1秒。
    """
    start_time = time.time()
    remaining = seconds
    while remaining > 0:
        sleep_time = min(remaining, check_interval)
        time.sleep(sleep_time)
        remaining = seconds - (time.time() - start_time)

def go(name) -> Optional[GameObject]:
    """
    指定された名前のゲームオブジェクトを検索して返します。

    Args:
        name (str): 検索するゲームオブジェクトの名前

    Returns:
        GameObject: 見つかったゲームオブジェクト。見つからない場合はNone。
    """
    return game_object.find(name)

def comment(text):
    """
    指定されたテキストをエージェントのコメントとして表示します。

    Args:
        text (str): 表示するコメントテキスト
    """
    try:
        rpc.invoke('Comment', text)
    except RPCError as e:
        if e.error.code == RPCErrorCode.MethodNotFound:
            error_log('Comment is not supported in this agent')
        else:
            raise e

def ask(question: str, answer_type: str = 'str') -> str|int|bool:
    """
    指定された質問と現在のゲームのスクリーンショットをエージェントに投げかけ、回答を取得します。

    Args:
        question (str): 質問テキスト
        answer_type (str, optional): 期待される回答の型（'str', 'int', 'bool'）。デフォルトは'str'。

    Returns:
        str|int|bool: エージェントからの回答
    """
    return respondent.ask(question, answer_type)

@contextmanager
def timeout_override(timeout_seconds: int):
    """
    オートプレイのタイムアウトを一時的に上書きします。
    withブロックを抜けると自動的にタイムアウトがクリアされます。

    使用例:
        >>> with timeout_override(30):
        ...     # ここでタイムアウトが30秒に設定される
        ...     tap("button")
        ... # withブロックを抜けると自動的にタイムアウトがクリアされる

    Args:
        timeout_seconds (int): タイムアウト秒数
    """
    set_timeout_override(timeout_seconds)
    try:
        yield
    finally:
        clear_timeout_override()

def set_value(key: str, value: str | bool):
    """
    オートプレイ実行時の変数を設定します。

    Args:
        key (str): 変数のキー
        value (str | bool): 設定する値
    """
    if isinstance(value, bool):
        rpc.invoke('SetBoolValue', key, value)
    elif isinstance(value, str):
        rpc.invoke('SetStringValue', key, value)
    else:
        error_log(f"Unsupported value type for set_value: {type(value).__name__}. Only str and bool are supported.")
