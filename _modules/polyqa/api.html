def get_latest_screenshot() -> ScreenshotFrame:
    screenshot = rpc.invoke('GetLatestScreenshot')
    if not screenshot['Data']:
        raise ValueError('No screenshot')
    screenshot_bytes = base64.b64decode(screenshot['Data'])
    return ScreenshotFrame(screenshot['Frame'], Image.open(io.BytesIO(screenshot_bytes)))

def get_screenshot(frame: int) -> Image.Image:
    screenshot = rpc.invoke('GetScreenshot', frame)
    if not screenshot:
        raise ValueError('No screenshot')
    screenshot_bytes = base64.b64decode(screenshot)
    return Image.open(io.BytesIO(screenshot_bytes))

def get_user_id() -> str:
    licensee_id = rpc.invoke('GetLicenseeId')
    user_id = rpc.invoke('GetUserId')
    return f'{licensee_id}/{user_id}'

def get_action_log(start: Optional[datetime]) -> list[ActionLog]:
    start_time = start.isoformat() if start else None
    actions = rpc.invoke('GetActionLog', start_time)
    return [ActionLog(action) for action in actions]
