class GameObject(UnityObject):
    """
    ゲームオブジェクトのラッパークラス。

    """



[ドキュメント]
    def find(self, name=None, pattern=None) -> Optional['GameObject']:
        """
        現在のゲームオブジェクト内で指定された名前またはパターンに一致する最初のゲームオブジェクトを検索します。

        Args:
            name (str, optional): 検索するゲームオブジェクトの名前
            pattern (str, optional): 検索するゲームオブジェクトの正規表現パターン

        Returns:
            int: 見つかったゲームオブジェクト。見つからない場合はNone。

        Raises:
            ValueError: nameもpatternも指定されていない場合
        """
        if name is not None:
            instance_id = rpc.invoke('FindGameObject', self.id, name)
        elif pattern is not None:
            instance_id = rpc.invoke('MatchGameObject', self.id, pattern)
        else:
            raise ValueError('name or pattern must be specified')
        if instance_id == 0:
            return None
        return GameObject(instance_id)

    def find_all(self, name=None, pattern=None) -> list['GameObject']:
        """
        現在のゲームオブジェクト内で指定された名前またはパターンに一致するすべてのゲームオブジェクトを検索します。

        Args:
            name (str, optional): 検索するゲームオブジェクトの名前
            pattern (str, optional): 検索するゲームオブジェクトの正規表現パターン

        Returns:
            list[GameObject]: 見つかったゲームオブジェクトのリスト

        Raises:
            ValueError: nameもpatternも指定されていない場合
        """
        if name is not None:
            ids = rpc.invoke('FindGameObjects', self.id, name)
        elif pattern is not None:
            ids = rpc.invoke('MatchGameObjects', self.id, pattern)
        else:
            raise ValueError('name or pattern must be specified')
        return [GameObject(i) for i in ids]

def find(name) -> Optional[GameObject]:
    """
    指定された名前のゲームオブジェクトを検索します。

    Args:
        name (str): 検索するゲームオブジェクトの名前

    Returns:
        GameObject: 見つかったゲームオブジェクト。見つからない場合はNone。
    """
