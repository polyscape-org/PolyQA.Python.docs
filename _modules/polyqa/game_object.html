    def find(self, name=None, pattern=None):
        """
        現在のゲームオブジェクト内で指定された名前またはパターンに一致する最初のゲームオブジェクトを検索します。

        Args:
            name (str, optional): 検索するゲームオブジェクトの名前
            pattern (str, optional): 検索するゲームオブジェクトの正規表現パターン

        Returns:
            GameObject: 見つかったゲームオブジェクト。見つからない場合はNone。

        Raises:
            ValueError: nameもpatternも指定されていない場合
        """
        if name is not None:
            go = self.game_object.FindGameObject(name)
        elif pattern is not None:
            go = self.game_object.MatchGameObject(pattern)
        else:
            raise ValueError('name or pattern must be specified')
        if go is None:
            return None
        return GameObject(go)

    def find_all(self, name=None, pattern=None):
        """
        現在のゲームオブジェクト内で指定された名前またはパターンに一致するすべてのゲームオブジェクトを検索します。

        Args:
            name (str, optional): 検索するゲームオブジェクトの名前
            pattern (str, optional): 検索するゲームオブジェクトの正規表現パターン

        Returns:
            list[GameObject]: 見つかったゲームオブジェクトのリスト

        Raises:
            ValueError: nameもpatternも指定されていない場合
        """
        if name is not None:
            goes = self.game_object.FindGameObjects(name)
        elif pattern is not None:
            goes = self.game_object.MatchGameObjects(pattern)
        else:
            raise ValueError('name or pattern must be specified')
        return [GameObject(go) for go in goes]

def find(name) -> Optional[GameObject]:
    """
    ルートから指定された名前のゲームオブジェクトを検索します。

    Args:
        name (str): 検索するゲームオブジェクトの名前

    Returns:
        GameObject: 見つかったゲームオブジェクト。見つからない場合はNone。
    """
    go = Context.skill.FindGameObject(name)
    if go is None:
        return None
    return GameObject(go)
